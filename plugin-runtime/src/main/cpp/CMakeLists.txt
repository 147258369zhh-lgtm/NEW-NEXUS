cmake_minimum_required(VERSION 3.22.1)
project(nexus_plugin_runtime LANGUAGES C CXX)
option(NEXUS_WASM_HOST_CHECK "Build the same WASM bridge for a local host check" OFF)

if(NOT NEXUS_WASM_HOST_CHECK)
enable_language(ASM)
set(QUICKJS_DIR "${CMAKE_CURRENT_LIST_DIR}/quickjs")
file(READ "${QUICKJS_DIR}/VERSION" QUICKJS_VERSION_RAW)
string(STRIP "${QUICKJS_VERSION_RAW}" QUICKJS_VERSION)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/operit_quickjs.h" "#pragma once\n#include \"${QUICKJS_DIR}/quickjs.h\"\n")

add_library(quickjsjni SHARED quickjs_jni.cpp
    "${QUICKJS_DIR}/cutils.c" "${QUICKJS_DIR}/dtoa.c"
    "${QUICKJS_DIR}/libregexp.c" "${QUICKJS_DIR}/libunicode.c" "${QUICKJS_DIR}/quickjs.c")
target_compile_definitions(quickjsjni PRIVATE _GNU_SOURCE CONFIG_VERSION="${QUICKJS_VERSION}" NDEBUG)
target_compile_options(quickjsjni PRIVATE -O3 -fwrapv -funsigned-char -Wall -Wextra
    -Wno-sign-compare -Wno-missing-field-initializers -Wno-unused-parameter)
target_include_directories(quickjsjni PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(quickjsjni PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_link_libraries(quickjsjni android log m)
target_link_options(quickjsjni PRIVATE "-Wl,-z,max-page-size=16384")
endif()

set(WAMR_ROOT_DIR "${CMAKE_CURRENT_LIST_DIR}/wamr")
if(NEXUS_WASM_HOST_CHECK)
    set(WAMR_BUILD_PLATFORM "windows")
    set(WAMR_BUILD_TARGET "X86_64")
else()
    set(WAMR_BUILD_PLATFORM "android")
    set(WAMR_BUILD_TARGET "AARCH64")
endif()
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_LIBC_UVWASI 0)
set(WAMR_BUILD_MULTI_MODULE 0)
set(WAMR_BUILD_LIB_PTHREAD 0)
set(WAMR_BUILD_LIB_WASI_THREADS 0)
set(WAMR_BUILD_THREAD_MGR 0)
set(WAMR_BUILD_SHARED_HEAP 0)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_REF_TYPES 0)
set(WAMR_BUILD_GC 0)
set(WAMR_BUILD_INSTRUCTION_METERING 1)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
include("${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake")
add_library(nexus_wamr STATIC ${WAMR_RUNTIME_LIB_SOURCE})
target_include_directories(nexus_wamr PUBLIC "${WAMR_ROOT_DIR}/core/iwasm/include")
if(NOT MSVC)
    target_compile_options(nexus_wamr PRIVATE -O3 -g0)
endif()
set_target_properties(nexus_wamr PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(toolpkgwasm SHARED toolpkg_wasm_jni.cpp)
target_include_directories(toolpkgwasm PRIVATE "${WAMR_ROOT_DIR}/core/iwasm/include")
target_compile_features(toolpkgwasm PRIVATE cxx_std_17)
if(NEXUS_WASM_HOST_CHECK)
    find_package(JNI REQUIRED)
    target_include_directories(toolpkgwasm PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(toolpkgwasm nexus_wamr)
else()
    target_link_libraries(toolpkgwasm nexus_wamr android log m dl)
    target_link_options(toolpkgwasm PRIVATE "-Wl,-z,max-page-size=16384")
endif()
